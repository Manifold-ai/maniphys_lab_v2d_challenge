   11  def resample_times(num_frames: int, source_fps: float, target_fps: float) -> tuple[np.ndarray, np.ndarray]:
   12      """Return uniform source and target timestamps spanning the same motion duration."""
   13      _validate_timing(num_frames, source_fps, target_fps)
   14      duration = (num_frames - 1) / source_fps
   15      num_target_frames = int(round(duration * target_fps)) + 1
   16      source_times = np.arange(num_frames, dtype=np.float64) / source_fps
   17      target_times = np.clip(np.arange(num_target_frames, dtype=np.float64) / target_fps, 0.0, duration)
   18      return source_times, target_times
